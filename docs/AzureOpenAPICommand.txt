# -----------------------------------------------------------------------------
# Create an Azure OpenAI resource.
# Run this only once. The resource name must be globally unique.
# If the resource already exists, this command will fail.
# -----------------------------------------------------------------------------
az cognitiveservices account create \
  --name ai-pr-reviewer-openai-harshil \
  --resource-group ai-pr-reviewer-rg \
  --kind OpenAI \
  --sku S0 \
  --location eastus \
  --yes


# -----------------------------------------------------------------------------
# Create a GPT-5 Mini deployment inside the Azure OpenAI resource.
#
# --deployment-name : The deployment name your application uses
#                     (AZURE_OPENAI_DEPLOYMENT).
# --model-name      : The OpenAI model to deploy.
# --model-version   : Specific model version.
# --sku-capacity    : Quota units (TPM allocation) assigned to this deployment.
#
# NOTE:
# - This command fails if the deployment already exists.
# - It also fails if your subscription doesn't have enough available quota.
# -----------------------------------------------------------------------------
az cognitiveservices account deployment create \
  --name ai-pr-reviewer-openai-harshil \
  --resource-group ai-pr-reviewer-rg \
  --deployment-name gpt-5-mini \
  --model-name gpt-5-mini \
  --model-version 2025-08-07 \
  --model-format OpenAI \
  --sku-name GlobalStandard \
  --sku-capacity 50


# -----------------------------------------------------------------------------
# Get the Azure OpenAI endpoint.
# Use this value as AZURE_OPENAI_ENDPOINT.
# Example:
# https://ai-pr-reviewer-openai-harshil.openai.azure.com/
# -----------------------------------------------------------------------------
az cognitiveservices account show \
  --name ai-pr-reviewer-openai-harshil \
  --resource-group ai-pr-reviewer-rg \
  --query properties.endpoint \
  --output tsv


# -----------------------------------------------------------------------------
# Get one of the API keys for the Azure OpenAI resource.
# Use this value as AZURE_OPENAI_API_KEY.
# -----------------------------------------------------------------------------
az cognitiveservices account keys list \
  --name ai-pr-reviewer-openai-harshil \
  --resource-group ai-pr-reviewer-rg \
  --query key1 \
  --output tsv


# -----------------------------------------------------------------------------
# List deployment names only.
# Useful for verifying that the deployment exists.
# -----------------------------------------------------------------------------
az cognitiveservices account deployment list \
  --name ai-pr-reviewer-openai-harshil \
  --resource-group ai-pr-reviewer-rg \
  --query "[].name" \
  --output tsv


# -----------------------------------------------------------------------------
# List all deployments in a table format.
# Shows deployment name, model, SKU, and other details.
# -----------------------------------------------------------------------------
az cognitiveservices account deployment list \
  --resource-group ai-pr-reviewer-rg \
  --name ai-pr-reviewer-openai-harshil \
  --output table


# -----------------------------------------------------------------------------
# Attempt to update an existing deployment's capacity.
#
# NOTE:
# Azure CLI currently DOES NOT support the 'deployment update' command.
# This command will fail with:
#   "'update' is misspelled or not recognized by the system."
#
# To change capacity, either:
#   1. Use Azure AI Foundry/Portal, or
#   2. Delete and recreate the deployment.
# -----------------------------------------------------------------------------
az cognitiveservices account deployment update \
  --resource-group ai-pr-reviewer-rg \
  --name ai-pr-reviewer-openai-harshil \
  --deployment-name gpt-5-mini \
  --sku-capacity 50


# -----------------------------------------------------------------------------
# Recreate the deployment with a larger quota.
#
# IMPORTANT:
# Delete the existing deployment first, otherwise this command will fail because
# a deployment with the same name already exists.
#
# This command will also fail if your Azure subscription doesn't have enough
# available TPM quota.
# -----------------------------------------------------------------------------
az cognitiveservices account deployment create \
  --name ai-pr-reviewer-openai-harshil \
  --resource-group ai-pr-reviewer-rg \
  --deployment-name gpt-5-mini \
  --model-name gpt-5-mini \
  --model-version 2025-08-07 \
  --model-format OpenAI \
  --sku-name GlobalStandard \
  --sku-capacity 50
